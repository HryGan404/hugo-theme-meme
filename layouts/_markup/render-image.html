{{- $src := .Destination -}} {{- $resource := "" -}} {{- if not (findRE `^https?:` $src) -}} {{- if strings.HasPrefix $src "/" -}} {{- with resources.Get (strings.TrimPrefix "/" $src) -}} {{- $resource = . -}} {{- end -}} {{- else -}} {{- with .Page.Resources.GetMatch $src -}} {{- $resource = . -}} {{- end -}} {{- end -}} {{- end -}} {{- $width := .Attributes.width -}} {{- $height := .Attributes.height -}} {{- if and $resource (ne $resource.MediaType.SubType "svg") -}} {{- $width = $width | default $resource.Width -}} {{- $height = $height | default $resource.Height -}} {{- end -}}
